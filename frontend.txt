    <!-- UI Logic Controller -->
    <script>
        let wsClient = null;

        function establishGatewayConnection() {
            const token = document.getElementById('auth-token').value;
            if(!token) return alert("Please supply an execution token.");

            const protocol = window.location.protocol === 'https:' ? 'wss://' : 'ws://';
            const gatewayUrl = `${protocol}${window.location.host}?token=${encodeURIComponent(token)}`;

            wsClient = new WebSocket(gatewayUrl);

            wsClient.onopen = () => {
                document.getElementById('login-overlay').classList.add('hidden');
                document.getElementById('dashboard').classList.remove('hidden');
            };

            wsClient.onmessage = (event) => {
                const packet = JSON.parse(event.data);
                const stream = document.getElementById('terminal-stream');

                if (packet.type === 'status') {
                    stream.innerHTML += `<div class="text-zinc-500 text-xs">⚠️ [SYSTEM] ${packet.message}</div>`;
                } else if (packet.type === 'text') {
                    stream.innerHTML += `<div class="bg-zinc-900 border border-zinc-800 p-3 rounded text-zinc-200 shadow-md">🤖 <span class="font-sans leading-relaxed">${packet.message}</span></div>`;
                } else if (packet.type === 'code') {
                    stream.innerHTML += `<div class="text-amber-500 text-xs font-bold">✨ Updated file context: ${packet.filename}</div>`;
                    document.getElementById('current-filename').innerText = packet.filename;
                    document.getElementById('code-canvas').innerText = packet.data;
                } else if (packet.type === 'error') {
                    stream.innerHTML += `<div class="text-red-500 text-xs font-bold">❌ [ERROR] ${packet.message}</div>`;
                }
                
                stream.scrollTop = stream.scrollHeight;
            };

            wsClient.onclose = () => {
                alert("🔒 Session closed. Connection interrupted or password invalid.");
                window.location.reload();
            };
        }

        function dispatchInstruction() {
            const input = document.getElementById('cmd-input');
            const data = input.value.trim();
            if(!data || !wsClient) return;

            // Render user command in local UI immediately
            const stream = document.getElementById('terminal-stream');
            stream.innerHTML += `<div class="text-right"><span class="inline-block bg-amber-500/10 border border-amber-500/20 text-amber-400 p-3 rounded text-left font-sans text-sm max-w-[85%]">👤 ${data}</span></div>`;
            
            wsClient.send(JSON.stringify({ type: 'command', data: data }));
            input.value = '';
            stream.scrollTop = stream.scrollHeight;
        }
    </script>
